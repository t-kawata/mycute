```rust
use std::fs;
use std::path::Path;
use parking_lot::RwLock;
use std::sync::Arc;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Settings {
    pub hotkeys: HotkeyConfig,
    pub filler_words: Vec<String>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct HotkeyConfig {
    pub start: Vec<String>, // e.g., ["Control", "KeyS"]
    pub commit: Vec<String>,
}

impl Default for Settings {
    fn default() -> Self {
        Self {
            hotkeys: HotkeyConfig {
                start: vec!["Option".to_string(), "KeyS".to_string()],
                commit: vec!["Option".to_string(), "KeyC".to_string()],
            },
            filler_words: vec!["えー".into(), "あのー".into(), "えっと".into()],
        }
    }
}

pub struct ConfigManager {
    pub config: Arc<RwLock<Settings>>,
}

impl ConfigManager {
    pub fn new() -> Self {
        let path = Path::new("settings.json");
        let settings = if path.exists() {
            let content = fs::read_to_string(path).expect("Read config failed");
            serde_json::from_str(&content).unwrap_or_default()
        } else {
            Settings::default()
        };
        Self { config: Arc::new(RwLock::new(settings)) }
    }
}
```

### 2.4. src/input/keyboard.rs
```rust
use objc2::{class, msg_send, sel, sel_impl};
use objc2_core_graphics::{CGEventCreateKeyboardEvent, CGEventPost, CGKeyCode, CGEventSourceRef};
use std::ptr;

pub struct KeyboardInjector;

impl KeyboardInjector {
    pub fn is_authorized() -> bool {
        unsafe {
            // AXIsProcessTrustedWithOptions を使うのが確実だが、簡易的に AXUIElementIsTrusted をチェック
            let trusted: bool = msg_send![class!(AXUIElement), isProcessTrusted];
            trusted
        }
    }

    pub fn type_text(text: &str) {
        for ch in text.chars() {
            if let Some(code) = Self::char_to_keycode(ch) {
                let shift = ch.is_uppercase() || "~!@#$%^&*()_+{}|:\"<>?".contains(ch);
                unsafe {
                    let source = ptr::null_mut() as CGEventSourceRef;
                    if shift { Self::post_key(56, true, source); } // Shift Down
                    Self::post_key(code, true, source);
                    Self::post_key(code, false, source);
                    if shift { Self::post_key(56, false, source); } // Shift Up
                }
            }
            std::thread::sleep(std::time::Duration::from_millis(5));
        }
    }

    unsafe fn post_key(code: CGKeyCode, down: bool, source: CGEventSourceRef) {
        let event = CGEventCreateKeyboardEvent(source, code, down);
        CGEventPost(1, event); // kCGHIDEventTap = 1
    }

    fn char_to_keycode(ch: char) -> Option<CGKeyCode> {
        match ch.to_ascii_lowercase() {
            'a' => Some(0x00), 'b' => Some(0x0B), 'c' => Some(0x08), 'd' => Some(0x02),
            'e' => Some(0x0E), 'f' => Some(0x03), 'g' => Some(0x05), 'h' => Some(0x04),
            'i' => Some(0x22), 'j' => Some(0x26), 'k' => Some(0x28), 'l' => Some(0x25),
            'm' => Some(0x2E), 'n' => Some(0x2D), 'o' => Some(0x1F), 'p' => Some(0x23),
            'q' => Some(0x0C), 'r' => Some(0x0F), 's' => Some(0x01), 't' => Some(0x11),
            'u' => Some(0x20), 'v' => Some(0x09), 'w' => Some(0x0D), 'x' => Some(0x07),
            'y' => Some(0x10), 'z' => Some(0x06),
            '0' => Some(0x1D), '1' => Some(0x12), '2' => Some(0x13), '3' => Some(0x14),
            '4' => Some(0x15), '5' => Some(0x17), '6' => Some(0x16), '7' => Some(0x1A),
            '8' => Some(0x1C), '9' => Some(0x19),
            ' ' => Some(0x31), '.' => Some(0x2F), ',' => Some(0x2B), '\n' => Some(0x24),
            _ => None,
        }
    }
}
```

### 2.5. src/ui/popup.rs
```rust
use objc2::{class, msg_send, sel, sel_impl};
use objc2_foundation::{NSRect, NSPoint, NSSize, NSString, MainThreadMarker};
use objc2_app_kit::{NSWindow, NSTextView, NSWindowStyleMask, NSBackingStoreType};
use tokio::sync::{mpsc, oneshot};
use std::sync::Arc;
use parking_lot::Mutex;

pub enum UiCommand {
    UpdateText(String),
    GetText(oneshot::Sender<String>),
    Show,
    Hide,
}

pub struct PopupWindow {
    tx: mpsc::UnboundedSender<UiCommand>,
}

impl PopupWindow {
    pub fn new() -> Self {
        let (tx, mut rx) = mpsc::unbounded_channel::<UiCommand>();
        
        // AppKit は Main Thread で動作させる必要があるため、
        // 実際には main.rs のメインループでこれを捌くか、dispatch_get_main_queue 相当を行う。
        // ここでは単純化のため、専用スレッドで RunLoop を回す。
        std::thread::spawn(move || {
            autoreleasepool(|_| {
                unsafe {
                    let frame = NSRect::new(NSPoint::new(100.0, 100.0), NSSize::new(400.0, 150.0));
                    let style = NSWindowStyleMask::Borderless;
                    let window: *mut NSWindow = msg_send![class!(NSWindow), alloc];
                    let window: *mut NSWindow = msg_send![window, initWithContentRect:frame
                        styleMask:style
                        backing:NSBackingStoreType::Buffered
                        defer:false];
                    
                    let tv_frame = NSRect::new(NSPoint::new(0.0, 0.0), NSSize::new(400.0, 150.0));
                    let tv: *mut NSTextView = msg_send![class!(NSTextView), alloc];
                    let tv: *mut NSTextView = msg_send![tv, initWithFrame:tv_frame];
                    
                    let content_view: *mut objc2::runtime::Object = msg_send![window, contentView];
                    let _: () = msg_send![content_view, addSubview:tv];
                    let _: () = msg_send![window, setLevel: 3]; // Floating
                    
                    while let Some(cmd) = rx.blocking_recv() {
                        match cmd {
                            UiCommand::UpdateText(text) => {
                                let ns_text = NSString::from_str(&text);
                                let _: () = msg_send![tv, setString:&ns_text];
                            }
                            UiCommand::GetText(reply) => {
                                let ns_string: *mut NSString = msg_send![tv, string];
                                let mut result = String::new();
                                if !ns_string.is_null() {
                                    result = ns_string.as_ref().unwrap().to_string();
                                }
                                let _ = reply.send(result);
                            }
                            UiCommand::Show => {
                                let _: () = msg_send![window, makeKeyAndOrderFront: ptr::null_mut::<()>()];
                            }
                            UiCommand::Hide => {
                                let _: () = msg_send![window, orderOut: ptr::null_mut::<()>()];
                            }
                        }
                    }
                }
            });
        });

        Self { tx }
    }

    pub fn update(&self, text: String) { let _ = self.tx.send(UiCommand::UpdateText(text)); }
    pub fn show(&self) { let _ = self.tx.send(UiCommand::Show); }
    pub fn hide(&self) { let _ = self.tx.send(UiCommand::Hide); }
    pub async fn get_text(&self) -> String {
        let (tx, rx) = oneshot::channel();
        let _ = self.tx.send(UiCommand::GetText(tx));
        rx.await.unwrap_or_default()
    }
}

// 簡易的な autoreleasepool の代替 (objc2-foundation にあるものを使用)
fn autoreleasepool<F, R>(f: F) -> R where F: FnOnce(&objc2_foundation::NSAutoreleasePool) -> R {
    let pool = unsafe { objc2_foundation::NSAutoreleasePool::new() };
    let res = f(&pool);
    res
}
```

### 2.7. src/main.rs
```rust
mod types;
mod config;
mod hotkey;
mod input;
mod ui;
mod stt;

use types::{AppState, HotkeyAction, SttEvent};
use config::ConfigManager;
use input::keyboard::KeyboardInjector;
use ui::popup::PopupWindow;
use tokio::sync::mpsc;

#[tokio::main]
async fn main() {
    env_logger::init();
    
    if !KeyboardInjector::is_authorized() {
        eprintln!("【エラー】アクセシビリティ権限がありません。システム設定で許可してください。");
        return;
    }

    let config_mgr = ConfigManager::new();
    let popup = PopupWindow::new();
    let (stt_tx, mut stt_rx) = mpsc::channel::<SttEvent>(100);
    let mut state = AppState::Idle;

    // ホットキー監視開始 (rdev)
    let (hk_tx, mut hk_rx) = mpsc::channel::<HotkeyAction>(10);
    let config_clone = config_mgr.config.read().clone();
    std::thread::spawn(move || {
        // rdev の listen ループ
        // 省略: config_clone に基づいた判定
    });

    println!("Tahoe Voice Tool Started.");

    loop {
        tokio::select! {
            Some(action) = hk_rx.recv() => {
                match action {
                    HotkeyAction::Start => {
                        if state == AppState::Idle {
                            popup.show();
                            state = AppState::Recording;
                        }
                    }
                    HotkeyAction::Commit => {
                        if state == AppState::Recording {
                            let text = popup.get_text().await;
                            KeyboardInjector::type_text(&text);
                            popup.hide();
                            state = AppState::Idle;
                        }
                    }
                }
            }
            Some(event) = stt_rx.recv() => {
                match event {
                    SttEvent::PartialResult(t) => popup.update(t),
                    SttEvent::FinalResult(t) => popup.update(t),
                    SttEvent::Error(e) => eprintln!("STT Error: {}", e),
                }
            }
        }
    }
}
```
